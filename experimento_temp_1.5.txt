#include <stdio.h>
#include <stdlib.h>

// Função de troca de elementos
void trocar(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

// Função de Bubble Sort
void bubbleSort(int arr[], int n) {
    int i, j;
    // O algoritmo percorre a lista por n vezes
    for (i = 0; i < n-1; i++) {
        // O último i elementos já estão ordenados, então a comparação pode ser feita apenas até n-i-1
        for (j = 0; j < n-i-1; j++) {
            // Se o elemento encontrado for maior que o próximo, trocar os dois
            if (arr[j] > arr[j+1]) {
                trocar(&arr[j], &arr[j+1]);
            }
        }
    }
}

// Função para imprimir o array
void imprimirArray(int arr[], int size) {
    int i;
    for (i=0; i < size; i++)
        printf("%d ", arr[i]);
    printf("\n");
}
