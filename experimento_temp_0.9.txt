#include <stdio.h>

// Função que implementa o Bubble Sort
void bubbleSort(int arr[], int n) {
    int i, j, temp;
    int trocas = 0, comparações = 0;

    // Exibir informações iniciais
    printf("Bubble Sort em andamento...\n");
    printf("Tamanho do vetor: %d\n", n);

    // Loop externo, controla o número de passos
    for (i = 0; i < n-1; i++) {
        // Flag para indicar se houve trocas
        int houveTroca = 0;

        // Loop interno, compara adjacentes e ordena
        for (j = 0; j < n-i-1; j++) {
            comparações++;

            // Se o elemento atual for maior que o próximo, troca-os
            if (arr[j] > arr[j+1]) {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
                houveTroca = 1;
                trocas++;
            }
        }

        // Se não houve trocas nesta passagem, o vetor já está ordenado
        if (!houveTroca) {
            break;
        }
    }
